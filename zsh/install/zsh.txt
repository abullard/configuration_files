# ========= START CONFIGURATION_FILES SETUP =========

# --- used for starship inspiration
export INSPIRE_LINE="$(shuf -n 1 $HOME/Developer/configuration_files/zsh/assets/inspiration.txt)"

# --- Setup Z
if echo $(uname -s) | grep -q "Linux"; then
    source $HOME/.z.sh
elif echo $(uname -s) | grep -q "Darwin"; then
    source /opt/homebrew/etc/profile.d/z.sh
else
    echo "ERROR: z is not properly sourced!!"
fi

# --- unalias existing commands
unalias z 2> /dev/null
unalias env 2> /dev/null

# --- source all custom bash scripts for the terminal
if echo $(uname -s) | grep -q "Linux"; then
    ZSH_FUNC_PATH="$HOME/Developer/configuration_files/zsh/functions/linux"
elif echo $(uname -s) | grep -q "Darwin"; then
    ZSH_FUNC_PATH="$HOME/Developer/configuration_files/zsh/functions/macos"
else
    echo "ERROR: Cannot find configuration_file functions to source!"
fi
FILES=($(ls $ZSH_FUNC_PATH))
for FILE in $FILES; do 
    source $ZSH_FUNC_PATH/$FILE
done

# --- alias mapping
alias z="my_z"
alias pushz="pushz"
alias branch="branch"
alias deck="deck"
alias vimp="vimp"
alias guid="guid"
alias env="my_env"
alias lssh="my_ssh"
alias pal="pal"
# alias fls="fls" # needs some fine tuning, or deleted

# - bash
alias ll="ls -lha"
alias guid="guid"
alias c="clear"
alias accio="alias | grep"
alias editsh="vim $HOME/.zshrc"
alias editprofile="vim $HOME/.zprofile"
alias reload="echo 'reloading...'; source $HOME/.zprofile; source $HOME/.zshrc; clear"

# - npm/pnpm
alias npmview="npm view"
alias npmlist="npm list --depth=0"
alias akio="clear; npx @abullard/akio -d"

# - git
alias cleargitcreds="git credential-cache exit"
alias gs-="git switch -"
alias gcm="git commit -m"
alias gpu="git push"
alias changes='function _changes(){ git diff $1^1 $1 > changes.diff };_changes'

# --- fzf setup
source <(fzf --zsh)

# --- PS1 for default prompt styling (if needed)
export PS1="%F{yellow}%D{%Y-%m-%d}%fT%F{green}%*%f: %F{blue}^%3d%f %F{red}$%f "

# --- setup starship TUX prompt formatting (overwrites above PS1 formatting)
eval "$(starship init zsh)"

# ========= END CONFIGURATION_FILES SETUP =========
